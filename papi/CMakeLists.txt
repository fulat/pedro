cmake_minimum_required(VERSION 3.16)

project(pedro_papi VERSION 0.1 LANGUAGES CXX)


# -----------------------------------------------------------------------------
# Dependencies
# -----------------------------------------------------------------------------

find_package(
    Qt6
    REQUIRED
    COMPONENTS
        Core
        DBus
        Qml
)

find_package(
    tomlplusplus
    REQUIRED
)

find_package(
    PkgConfig
    REQUIRED
)

# GIO supplies installed desktop application metadata and GNOME favorites.
pkg_check_modules(
    GIO
    REQUIRED
    IMPORTED_TARGET
    gio-2.0
    gio-unix-2.0
)


# -----------------------------------------------------------------------------
# Sources
#
# Automatically discovers every C/C++ public header and source file inside PAPI.
#
# Example:
#
# papi/
# ├── io/
# │   ├── bluetooth/
# │   │   ├── manager.hpp
# │   │   └── manager.cpp
# │   │
# │   ├── network/
# │   │   └── wifi/
# │   │       ├── manager.hpp
# │   │       └── manager.cpp
# │   │
# │   └── fs/
# │       ├── fs.hpp
# │       └── fs.cpp
# │
# ├── gui/
# ├── system/
# └── utils/
#
# No file needs to be manually registered here.
# -----------------------------------------------------------------------------

file(
    GLOB_RECURSE
    PEDRO_PAPI_PUBLIC_HEADERS

    CONFIGURE_DEPENDS
    RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"

    "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp"
)

file(
    GLOB_RECURSE
    PEDRO_PAPI_SOURCES

    CONFIGURE_DEPENDS
    RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"

    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)


# -----------------------------------------------------------------------------
# Ignore build/generated directories
#
# This prevents files generated by CMake, CLion, etc. from accidentally
# becoming part of PAPI if a build directory exists inside the source tree.
# -----------------------------------------------------------------------------

list(
    FILTER PEDRO_PAPI_PUBLIC_HEADERS
    EXCLUDE
    REGEX "^(build|cmake-build[^/]*)/"
)

list(
    FILTER PEDRO_PAPI_SOURCES
    EXCLUDE
    REGEX "^(build|cmake-build[^/]*)/"
)


# -----------------------------------------------------------------------------
# Public include tree
#
# The source hierarchy is mirrored automatically under:
#
# build/.../papi/include/pedro/papi/
#
# Example:
#
# Source:
#
# io/bluetooth/manager.hpp
#
# Generated:
#
# include/
# └── pedro/
#     └── papi/
#         └── io/
#             └── bluetooth/
#                 └── manager.hpp
#
# Which allows:
#
# #include <pedro/papi/io/bluetooth/manager.hpp>
# -----------------------------------------------------------------------------

set(
    PEDRO_PAPI_GENERATED_INCLUDE
    "${CMAKE_CURRENT_BINARY_DIR}/include"
)

foreach(header IN LISTS PEDRO_PAPI_PUBLIC_HEADERS)

    get_filename_component(
        headerDirectory
        "${header}"
        DIRECTORY
    )

    file(
        MAKE_DIRECTORY
        "${PEDRO_PAPI_GENERATED_INCLUDE}/pedro/papi/${headerDirectory}"
    )

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/${header}"
        "${PEDRO_PAPI_GENERATED_INCLUDE}/pedro/papi/${header}"
        COPYONLY
    )

endforeach()


# -----------------------------------------------------------------------------
# Library
# -----------------------------------------------------------------------------

add_library(
    pedro_papi
    STATIC

    ${PEDRO_PAPI_SOURCES}
    ${PEDRO_PAPI_PUBLIC_HEADERS}
)

set_target_properties(pedro_papi PROPERTIES AUTOMOC ON)

target_compile_features(
    pedro_papi
    PUBLIC
        cxx_std_17
)


# -----------------------------------------------------------------------------
# Include directories
# -----------------------------------------------------------------------------

target_include_directories(
    pedro_papi

    PUBLIC
        $<BUILD_INTERFACE:${PEDRO_PAPI_GENERATED_INCLUDE}>
        $<INSTALL_INTERFACE:usr/include>

    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)


# -----------------------------------------------------------------------------
# Libraries
# -----------------------------------------------------------------------------

target_link_libraries(
    pedro_papi

    PUBLIC
        Qt6::Core
        Qt6::Qml

    PRIVATE
        Qt6::DBus
        PkgConfig::GIO
        tomlplusplus::tomlplusplus
)


# -----------------------------------------------------------------------------
# Install
# -----------------------------------------------------------------------------

install(
    TARGETS pedro_papi
    ARCHIVE DESTINATION usr/lib
)

install(
    DIRECTORY
        "${PEDRO_PAPI_GENERATED_INCLUDE}/"

    DESTINATION
        usr/include
)
